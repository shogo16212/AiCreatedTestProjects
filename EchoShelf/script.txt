```sql
/*============================================================
 Users
=============================================================*/
CREATE TABLE Users
(
    UserId              INT IDENTITY(1,1) PRIMARY KEY,
    UserName            NVARCHAR(50) NOT NULL,
    Email               NVARCHAR(255) NOT NULL UNIQUE,
    PasswordHash        NVARCHAR(255) NOT NULL,
    AvatarImagePath     NVARCHAR(300) NULL,

    CreatedAt           DATETIME2 NOT NULL DEFAULT(GETDATE()),
    UpdatedAt           DATETIME2 NULL,
    IsDeleted           BIT NOT NULL DEFAULT(0)
);
GO

/*============================================================
 Categories
=============================================================*/
CREATE TABLE Categories
(
    CategoryId          INT IDENTITY(1,1) PRIMARY KEY,
    CategoryName        NVARCHAR(50) NOT NULL UNIQUE,

    DisplayOrder        INT NOT NULL DEFAULT(0),

    CreatedAt           DATETIME2 NOT NULL DEFAULT(GETDATE()),
    IsDeleted           BIT NOT NULL DEFAULT(0)
);
GO

/*============================================================
 Tags
=============================================================*/
CREATE TABLE Tags
(
    TagId               INT IDENTITY(1,1) PRIMARY KEY,

    TagName             NVARCHAR(50) NOT NULL UNIQUE,

    CreatedAt           DATETIME2 NOT NULL DEFAULT(GETDATE()),
    IsDeleted           BIT NOT NULL DEFAULT(0)
);
GO

/*============================================================
 Settings
=============================================================*/
CREATE TABLE Settings
(
    SettingId           INT IDENTITY(1,1) PRIMARY KEY,

    SettingKey          NVARCHAR(100) NOT NULL UNIQUE,
    SettingValue        NVARCHAR(300) NOT NULL,

    Description         NVARCHAR(300) NULL
);
GO

/*============================================================
 Index
=============================================================*/

CREATE INDEX IX_Users_Email
ON Users(Email);

CREATE INDEX IX_Category_Name
ON Categories(CategoryName);

CREATE INDEX IX_Tag_Name
ON Tags(TagName);

GO
```

```sql
/*============================================================
 Memories
=============================================================*/
CREATE TABLE Memories
(
    MemoryId            INT IDENTITY(1,1) PRIMARY KEY,

    UserId              INT NOT NULL,
    CategoryId          INT NOT NULL,

    Title               NVARCHAR(100) NOT NULL,
    Episode             NVARCHAR(1000) NOT NULL,

    MemoryDate          DATE NOT NULL,

    IsFavorite          BIT NOT NULL DEFAULT(0),

    CreatedAt           DATETIME2 NOT NULL DEFAULT(GETDATE()),
    UpdatedAt           DATETIME2 NULL,

    IsDeleted           BIT NOT NULL DEFAULT(0),

    CONSTRAINT FK_Memories_User
        FOREIGN KEY(UserId)
        REFERENCES Users(UserId),

    CONSTRAINT FK_Memories_Category
        FOREIGN KEY(CategoryId)
        REFERENCES Categories(CategoryId)
);
GO

/*============================================================
 MemoryImages
=============================================================*/
CREATE TABLE MemoryImages
(
    ImageId             INT IDENTITY(1,1) PRIMARY KEY,

    MemoryId            INT NOT NULL,

    FileName            NVARCHAR(255) NOT NULL,
    FilePath            NVARCHAR(500) NOT NULL,

    FileSize            BIGINT NULL,

    CreatedAt           DATETIME2 NOT NULL DEFAULT(GETDATE()),

    CONSTRAINT FK_MemoryImages_Memory
        FOREIGN KEY(MemoryId)
        REFERENCES Memories(MemoryId)
        ON DELETE CASCADE
);
GO

/*============================================================
 MemoryTags
=============================================================*/
CREATE TABLE MemoryTags
(
    MemoryTagId         INT IDENTITY(1,1) PRIMARY KEY,

    MemoryId            INT NOT NULL,
    TagId               INT NOT NULL,

    CreatedAt           DATETIME2 NOT NULL DEFAULT(GETDATE()),

    CONSTRAINT FK_MemoryTags_Memory
        FOREIGN KEY(MemoryId)
        REFERENCES Memories(MemoryId)
        ON DELETE CASCADE,

    CONSTRAINT FK_MemoryTags_Tag
        FOREIGN KEY(TagId)
        REFERENCES Tags(TagId),

    CONSTRAINT UQ_Memory_Tag
        UNIQUE
        (
            MemoryId,
            TagId
        )
);
GO

/*============================================================
 INDEX
=============================================================*/

CREATE INDEX IX_Memories_User

CREATE INDEX IX_MemoryImages_Memory
ON MemoryImages(MemoryId);

CREATE INDEX IX_MemoryTags_Memory
ON MemoryTags(MemoryId);

CREATE INDEX IX_MemoryTags_Tag
ON MemoryTags(TagId);

GO
```

```sql
=============================================================*/

(
    ShelfItemId         INT IDENTITY(1,1) PRIMARY KEY,

    ShelfId             INT NOT NULL,
    MemoryId            INT NOT NULL,

    DisplayOrder        INT NOT NULL DEFAULT(1),

    CreatedAt           DATETIME2 NOT NULL DEFAULT(GETDATE()),

    CONSTRAINT FK_ShelfItems_Shelf
        FOREIGN KEY(ShelfId)
        REFERENCES Shelves(ShelfId)
        ON DELETE CASCADE,

    CONSTRAINT FK_ShelfItems_Memory
        FOREIGN KEY(MemoryId)
        REFERENCES Memories(MemoryId)
        ON DELETE CASCADE,

    CONSTRAINT UQ_Shelf_Memory
        UNIQUE
        (

            MemoryId

/*============================================================
 EchoAnalyses
=============================================================*/

(
```

/*============================================================
 VIEW
=============================================================*/

CREATE VIEW vw_MemorySummary
AS
SELECT
    M.MemoryId,
    U.UserName,
    C.CategoryName,
    M.Title,
    M.MemoryDate,
    M.IsFavorite
FROM Memories M
INNER JOIN Users U
    ON M.UserId = U.UserId
INNER JOIN Categories C
    ON M.CategoryId = C.CategoryId
WHERE M.IsDeleted = 0;
GO

CREATE VIEW vw_ShelfSummary
AS
SELECT
    S.ShelfId,
    S.ShelfName,
    U.UserName,
    COUNT(SI.ShelfItemId) AS ItemCount
FROM Shelves S
LEFT JOIN ShelfItems SI
    ON S.ShelfId = SI.ShelfId
INNER JOIN Users U
    ON S.UserId = U.UserId
WHERE S.IsDeleted = 0
GROUP BY
    S.ShelfId,
    S.ShelfName,
    U.UserName;
GO

/*============================================================
 Stored Procedure
=============================================================*/

CREATE PROCEDURE usp_GetMemoriesByUser
(
    @UserId INT
)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        MemoryId,
        Title,
        MemoryDate,
        IsFavorite
    FROM Memories
    WHERE UserId = @UserId
      AND IsDeleted = 0
    ORDER BY MemoryDate DESC;
END;
GO

CREATE PROCEDURE usp_GetEchoAnalysis
(
    @UserId INT
)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        EA.AnalysisDate,
        EAD.Score,
        T.TagName,
        C.CategoryName,
        EAD.Comment
    FROM EchoAnalyses EA
    INNER JOIN EchoAnalysisDetails EAD
        ON EA.AnalysisId = EAD.AnalysisId
    LEFT JOIN Tags T
        ON EAD.TagId = T.TagId
    LEFT JOIN Categories C
        ON EAD.CategoryId = C.CategoryId
    WHERE EA.UserId = @UserId
    ORDER BY EA.AnalysisDate DESC;
END;
GO
